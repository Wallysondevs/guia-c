    for (int i = 0; i < argc; i++) printf("argv[%d] = %s\n", i, argv[i]);
    char nome[50];
    int idade;
    printf("Seu nome: ");
    if (!fgets(nome, sizeof nome, stdin)) return 1;
    nome[strcspn(nome, "\n")] = '\0';
    printf("Sua idade: ");
    if (scanf("%d", &idade) != 1) return 2;

Várias abas (.h, .c, .txt)

#include <stdio.h>
#include "conta.h"

int main(void) {
    FILE *f = fopen("notas.txt", "r");
    if (!f) { perror("notas.txt"); return 1; }
    int n, soma = 0, qtd = 0;
    while (fscanf(f, "%d", &n) == 1) { soma += n; qtd++; }
    fclose(f);
    printf("Média: %.1f\n", media(soma, qtd));
    return 0;
}

#ifndef CONTA_H
#define CONTA_H
double media(int soma, int qtd);
#endif

#include "conta.h"

double media(int soma, int qtd) {
    return qtd ? (double)soma / qtd : 0.0;
}
