C stdlib.h e stdio.h

Esta é a página de consulta rápida das duas bibliotecas que quase todo programa em C inclui. O <stdlib.h> ("standard library", biblioteca padrão) tem memória dinâmica, conversão de texto em número, números aleatórios, ordenação e o encerramento do programa. O <stdio.h> ("standard input/output", entrada e saída padrão) tem a tela, o teclado e os arquivos. Cada linha das tabelas leva à aula que explica a função com calma. Os formatos do printf e do scanf estão em printf e scanf.

Todas as saídas foram produzidas de verdade: no gcc 15.2 da máquina da Ana (Linux de 64 bits, pasta ~/c/ref, comando gcc -std=c23 -Wall -Wextra) e no editor do guia (Clang em WebAssembly). Quando os dois diferem, a tabela diz.

stdlib.h: memória dinâmica

A memória dinâmica é a que o programa pede durante a execução, do tamanho que precisar, no heap (aula Pilha e Heap). Toda memória pedida tem de ser devolvida com free.

#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int *v = malloc(3 * sizeof *v);
    if (v == NULL) {
        return 1;
    }
    v[0] = 10; v[1] = 20; v[2] = 30;
    int *maior = realloc(v, 5 * sizeof *v);
    if (maior == NULL) {
        free(v);
        return 1;
    }
    v = maior;
    v[3] = 40; v[4] = 50;
    printf("realloc: %d %d %d %d %d\n", v[0], v[1], v[2], v[3], v[4]);
    free(v);
    double *z = calloc(4, sizeof *z);
    if (z == NULL) {
        return 1;
    }
    printf("calloc: %.1f %.1f %.1f %.1f\n", z[0], z[1], z[2], z[3]);
    free(z);
    char *gigante = malloc(300u * 1024 * 1024);
    printf("malloc de 300 MB: %s\n", gigante ? "deu certo" : "NULL");
    free(gigante);
    void *alinhado = aligned_alloc(64, 128);
    printf("aligned_alloc(64, 128): múltiplo de 64? %s\n", (unsigned long)alinhado % 64 == 0 ? "sim" : "não");
    free(alinhado);
    free(NULL);
    printf("free(NULL): não faz nada\n");
    return 0;
}

stdlib.h: texto → número

#include <stdio.h>
#include <stdlib.h>
#include <errno.h>

int main(void) {
    printf("atoi(\"42abc\") = %d\n", atoi("42abc"));
    printf("atoi(\"abc\")   = %d\n", atoi("abc"));
    printf("atof(\"3.14\")  = %g\n", atof("3.14"));
    char *fim;
    long n = strtol("  -17 kg", &fim, 10);
    printf("strtol: %ld, sobrou [%s]\n", n, fim);
    printf("strtol base 16: %ld\n", strtol("ff", NULL, 16));
    printf("strtol base 2:  %ld\n", strtol("1010", NULL, 2));
    printf("strtol base 0:  %ld %ld %ld\n", strtol("0x1F", NULL, 0), strtol("017", NULL, 0), strtol("17", NULL, 0));
    n = strtol("abc", &fim, 10);
    printf("strtol(\"abc\"): %ld, nada lido: %s\n", n, *fim == 'a' ? "sim" : "não");
    errno = 0;
    long long g = strtoll("99999999999999999999", NULL, 10);
    printf("strtoll grande demais: %lld, ERANGE: %s\n", g, errno == ERANGE ? "sim" : "não");
    unsigned long u = strtoul("4000000000", NULL, 10);
    printf("strtoul: %lu\n", u);
    double d = strtod("12.90 reais", &fim);
    printf("strtod: %.2f, sobrou [%s]\n", d, fim);
    float f = strtof("1e3", NULL);
    printf("strtof: %.1f\n", f);
    return 0;
}

Prefira as strto… às ato…: só elas dizem onde a leitura parou (fim) e se o número estourou (errno). Para um int, leia com strtol e confira se o valor está entre INT_MIN e INT_MAX (veja Tipos e Limites). O caminho inverso, número → texto, é o snprintf.

stdlib.h: ordenar, buscar e contas com inteiros

#include <stdio.h>
#include <stdlib.h>

int comparar(const void *a, const void *b) {
    int x = *(const int *)a, y = *(const int *)b;
    return (x > y) - (x < y);
}

int main(void) {
    printf("abs(-5) = %d, labs(-5L) = %ld, llabs(-5LL) = %lld\n", abs(-5), labs(-5L), llabs(-5LL));
    div_t r = div(17, 5);
    printf("div(17, 5): quot = %d, rem = %d\n", r.quot, r.rem);
    int notas[] = {7, 3, 9, 1, 5};
    qsort(notas, 5, sizeof notas[0], comparar);
    printf("qsort: %d %d %d %d %d\n", notas[0], notas[1], notas[2], notas[3], notas[4]);
    int chave = 7;
    int *achou = bsearch(&chave, notas, 5, sizeof notas[0], comparar);
    printf("bsearch(7): posição %td\n", achou - notas);
    chave = 4;
    printf("bsearch(4): %s\n", bsearch(&chave, notas, 5, sizeof notas[0], comparar) ? "achou" : "NULL");
    printf("getenv(\"HOME\") = %s\n", getenv("HOME"));
    printf("getenv(\"NAO_EXISTE\") = %s\n", getenv("NAO_EXISTE") ? "achou" : "NULL");
    printf("EXIT_SUCCESS = %d, EXIT_FAILURE = %d\n", EXIT_SUCCESS, EXIT_FAILURE);
    printf("RAND_MAX = %d\n", RAND_MAX);
    printf("MB_CUR_MAX = %zu\n", MB_CUR_MAX);
    return 0;
}

stdio.h: ler, escrever e andar no arquivo

#include <stdio.h>

int main(void) {
    FILE *f = fopen("compras.txt", "w");
    if (f == NULL) {
        perror("compras.txt");
        return 1;
    }
    fprintf(f, "arroz\nfeijão\n");
    fputs("café\n", f);
    fclose(f);

    f = fopen("compras.txt", "a");
    fputc('p', f);
    fputs("ão\n", f);
    fclose(f);

    f = fopen("compras.txt", "r");
    char linha[50];
    while (fgets(linha, sizeof linha, f) != NULL) {
        printf("lido: %s", linha);
    }
    printf("feof = %d, ferror = %d\n", feof(f) != 0, ferror(f) != 0);
    fseek(f, 0, SEEK_END);
    printf("tamanho: %ld bytes\n", ftell(f));
    rewind(f);
    int c = fgetc(f);
    ungetc(c, f);
    printf("primeira letra: %c, de novo: %c\n", c, fgetc(f));
    fclose(f);

    if (fopen("compras.txt", "wx") == NULL) {
        perror("modo \"wx\"");
    }
    if (rename("compras.txt", "lista.txt") == 0) {
        printf("rename: ok\n");
    }
    if (fopen("compras.txt", "r") == NULL) {
        perror("compras.txt");
    }
    printf("remove: %d\n", remove("lista.txt"));
    printf("remove de novo: %d\n", remove("lista.txt"));
    return 0;
}

O laço while (!feof(f)) leu a última nota duas vezes 
#include <stdio.h>

int main(void) {
    FILE *f = fopen("notas.txt", "w");
    if (f == NULL) {
        return 1;
    }
    fprintf(f, "7\n9\n");
    fclose(f);
    f = fopen("notas.txt", "r");
    if (f == NULL) {
        return 1;
    }
    int nota = 0, soma = 0;
    while (!feof(f)) {
        fscanf(f, "%d", &nota);
        printf("leu %d\n", nota);
        soma += nota;
    }
    printf("soma = %d\n", soma);
    fclose(f);
    return 0;
}

$ gcc -std=c23 -Wall -Wextra feof.c -o feof
$ ./feof
leu 7
leu 9
leu 9
soma = 25

O que significa: o feof não adivinha o futuro. Depois de ler o 9, ainda sobrou o \n, então o fim não foi alcançado. A volta seguinte chama fscanf, que falha (devolve EOF) e não muda nota: o 9 é somado de novo. Sem aviso nenhum, nos dois compiladores.

Como consertar: o teste é o retorno da leitura: while (fscanf(f, "%d", &nota) == 1), ou while (fgets(linha, sizeof linha, f) != NULL). É a missão abaixo. Aula: Ler Arquivos.

Não testou o NULL do fopen ou do malloc

Se o arquivo não existe, fopen devolve NULL, e o primeiro fgets(linha, 50, f) com esse f derruba o programa com Falha de segmentação (139), no gcc e no editor. Como consertar: if (f == NULL) { perror("compras.txt"); return 1; }, que mostra compras.txt: No such file or directory. Aula: errno e perror.

char c = getchar() num laço até EOF

O getchar devolve int porque o EOF (−1) precisa ser diferente de todos os 256 caracteres. Guardado num char, o EOF se confunde com o byte 255 (ou nunca é visto, se o char não tem sinal). Como consertar: int c;. Aula: char.

Experimente

Missão: o arquivo notas.txt (na outra aba) tem as notas 7 e 9, mas o programa soma 25, porque usa while (!feof(f)). Troque a condição do laço pelo retorno do fscanf (== 1), apague o fscanf de dentro do laço e faça o programa mostrar soma = 16.

#include <stdio.h>

int main(void) {
    FILE *f = fopen("notas.txt", "r");
    if (f == NULL) {
        perror("notas.txt");
        return 1;
    }
    int nota = 0, soma = 0;
    while (!feof(f)) {
        fscanf(f, "%d", &nota);
        printf("leu %d\n", nota);
        soma += nota;
    }
    printf("soma = %d\n", soma);
    fclose(f);
    return 0;
}

Resumo

  • stdlib.h: malloc/calloc/realloc/free (teste o NULL; todo malloc tem o seu free); strtol/strtod (melhores que atoi/atof); qsort/bsearch; abs/div; rand/srand; exit/atexit/abort; getenv/system.
  • stdio.h: printf/puts/putchar e scanf/fgets/getchar para tela e teclado; fopen ("r", "w" apaga, "a" acrescenta, "x" não apaga) e fclose; fread/fwrite para binário; fseek/ftell/rewind; remove/rename; perror.
  • Leia em laço pelo retorno da leitura, nunca com while (!feof(f)).
  • No editor, o long tem 4 bytes (o strtol estoura antes) e o malloc vai até 256 MB.
