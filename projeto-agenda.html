Projeto Agenda de Contatos

A Ana quer uma agenda no computador: digitar nome e telefone, listar todo mundo, achar um contato pelo nome (sem se importar com maiúsculas) e apagar quem saiu da vida dela. E, o mais importante, que a agenda não suma quando ela fechar o programa. É o projeto para juntar structs, funções, texto e arquivos num programa de verdade, com um menu que se repete.

Você vai praticar
  • Guardar nome e telefone juntos numa struct e ter um array de contatos.
  • Ler linhas com espaço usando fgets e tirar o \n do fim.
  • Um menu que se repete com while e switch (veja Sobre os Projetos).
  • Buscar sem diferenciar maiúsculas, passando o texto por tolower (ctype.h).
  • Remover um item de um array deslocando os seguintes para trás.
  • Gravar e ler um arquivo de texto com fprintf e sscanf, para a agenda sobreviver ao fechar o programa.

Antes de começar

Você precisa ter visto Structs, Funções, fgets e, para a versão 4, Arquivos. O arquivo vai se chamar agenda.c:

$ mkdir -p c/agenda
$ cd c/agenda
Por que ler tudo com fgets?

Nomes têm espaço ("Ana Paula"), e o scanf("%s") para no primeiro espaço (veja o Boletim de Notas). Por isso a agenda lê cada campo com fgets, que pega a linha inteira. Até a opção do menu é lida com fgets: assim não sobra nenhum \n atrapalhando a próxima leitura — o problema clássico de misturar scanf e fgets.

Versão 1 · A struct, o array e o menu

Primeiro, a base: uma struct Contato com nome e telefone, um array de contatos, e um menu que se repete com duas opções — adicionar e listar. Duas funções ajudam: tirar_novalinha (tira o \n que o fgets deixa) e ler_linha (mostra o rótulo, lê a linha e limpa o \n).

#include <stdio.h>
#include <string.h>

#define MAX_CONTATOS 100
#define TAM_NOME 50
#define TAM_TEL 20

struct Contato {
    char nome[TAM_NOME];
    char telefone[TAM_TEL];
};

void tirar_novalinha(char *s) {
    s[strcspn(s, "\n")] = '\0';
}

int ler_linha(const char *rotulo, char *destino, int tam) {
    printf("%s", rotulo);
    if (fgets(destino, tam, stdin) == NULL) {
        return 0;
    }
    tirar_novalinha(destino);
    return 1;
}

void adicionar(struct Contato agenda[], int *n) {
    struct Contato c = {0};
    ler_linha("Nome: ", c.nome, TAM_NOME);
    ler_linha("Telefone: ", c.telefone, TAM_TEL);
    agenda[*n] = c;
    (*n)++;
    printf("Contato adicionado.\n");
}

void listar(const struct Contato agenda[], int n) {
    if (n == 0) {
        printf("Agenda vazia.\n");
        return;
    }
    printf("\n%-20s %s\n", "Nome", "Telefone");
    for (int i = 0; i < n; i++) {
        printf("%-20s %s\n", agenda[i].nome, agenda[i].telefone);
    }
}

int main(void) {
    struct Contato agenda[MAX_CONTATOS];
    int n = 0;
    char opcao[10];
    int rodando = 1;
    while (rodando) {
        printf("\n=== Agenda da Ana ===\n");
        printf("1. Adicionar  2. Listar  0. Sair\n");
        printf("Opção: ");
        if (fgets(opcao, sizeof opcao, stdin) == NULL) {
            break;
        }
        switch (opcao[0]) {
            case '1': adicionar(agenda, &n); break;
            case '2': listar(agenda, n); break;
            case '0': rodando = 0; break;
            default: printf("Opção inválida.\n");
        }
    }
    printf("Até logo!\n");
    return 0;
}
struct Contato { char nome[TAM_NOME]; char telefone[TAM_TEL]; };
A "ficha" de um contato: nome (até 49 letras + o '\0') e telefone. Os tamanhos têm nome (#define), como manda as boas práticas.
s[strcspn(s, "\n")] = '\0';
O fgets guarda o Enter como \n no fim. O strcspn(s, "\n") dá a posição do \n (ou do fim); pôr um '\0' ali corta o Enter (veja fgets).
int ler_linha(const char *rotulo, char *destino, int tam)
Uma função só para ler um campo com segurança: mostra o rótulo, chama fgets com o tamanho do destino (nunca estoura, veja Segurança) e tira o \n. Devolve 0 se a entrada acabou.
agenda[*n] = c; (*n)++;
Guarda o novo contato na primeira posição livre e aumenta a contagem. O n vem por ponteiro porque a função precisa mudar o n do main.
printf("%-20s %s\n", ...)
%-20s põe o nome numa coluna de 20, à esquerda, para a lista ficar alinhada (veja Formatos do printf).
switch (opcao[0])
A opção é lida como texto; opcao[0] é a primeira tecla ('1', '2', '0'). Comparar o caractere evita o problema do \n do scanf.

Versão 2 · Buscar sem diferenciar maiúsculas

A Ana digita "ana" e quer achar "Ana Paula". Para isso, a busca não pode diferenciar maiúsculas de minúsculas. A ideia: passar os dois textos para minúsculas (com tolower) e ver se um está dentro do outro com strstr (veja Funções de String). A função contem_ignorando_caso faz isso; a buscar a usa em cada contato.

#include <stdio.h>
#include <string.h>
#include <ctype.h>

#define MAX_CONTATOS 100
#define TAM_NOME 50
#define TAM_TEL 20

struct Contato {
    char nome[TAM_NOME];
    char telefone[TAM_TEL];
};

void tirar_novalinha(char *s) {
    s[strcspn(s, "\n")] = '\0';
}

int ler_linha(const char *rotulo, char *destino, int tam) {
    printf("%s", rotulo);
    if (fgets(destino, tam, stdin) == NULL) {
        return 0;
    }
    tirar_novalinha(destino);
    return 1;
}

int contem_ignorando_caso(const char *texto, const char *parte) {
    char t[TAM_NOME];
    char p[TAM_NOME];
    int i;
    for (i = 0; texto[i] != '\0' && i < TAM_NOME - 1; i++) {
        t[i] = (char) tolower((unsigned char) texto[i]);
    }
    t[i] = '\0';
    for (i = 0; parte[i] != '\0' && i < TAM_NOME - 1; i++) {
        p[i] = (char) tolower((unsigned char) parte[i]);
    }
    p[i] = '\0';
    return strstr(t, p) != NULL;
}

void adicionar(struct Contato agenda[], int *n) {
    struct Contato c = {0};
    ler_linha("Nome: ", c.nome, TAM_NOME);
    ler_linha("Telefone: ", c.telefone, TAM_TEL);
    agenda[*n] = c;
    (*n)++;
    printf("Contato adicionado.\n");
}

void buscar(const struct Contato agenda[], int n) {
    char alvo[TAM_NOME];
    ler_linha("Buscar: ", alvo, TAM_NOME);
    int achou = 0;
    for (int i = 0; i < n; i++) {
        if (contem_ignorando_caso(agenda[i].nome, alvo)) {
            printf("%-20s %s\n", agenda[i].nome, agenda[i].telefone);
            achou = 1;
        }
    }
    if (!achou) {
        printf("Nenhum contato com \"%s\".\n", alvo);
    }
}

int main(void) {
    struct Contato agenda[MAX_CONTATOS];
    int n = 0;
    char opcao[10];
    int rodando = 1;
    while (rodando) {
        printf("\n=== Agenda da Ana ===\n");
        printf("1. Adicionar  3. Buscar  0. Sair\n");

            case '0': rodando = 0; break;
            default: printf("Opção inválida.\n");
        }
    }
    printf("Até logo!\n");
    return 0;
}
tolower((unsigned char) texto[i])
Transforma a letra em minúscula. O (unsigned char) evita um problema com acentos e valores negativos (veja ctype.h).
strstr(t, p) != NULL
strstr ("string in string") procura o texto p dentro de t e devolve onde ele começa, ou NULL se não achar. Assim a busca acha parte do nome: "cost" acha "Bruno Costa".
if (contem_ignorando_caso(agenda[i].nome, alvo))
Para cada contato, pergunta "o nome contém o que a Ana procura?". Se sim, mostra e marca achou = 1.

Versão 3 · Remover um contato

Remover de um array é mais sutil do que parece: não basta "apagar" a posição, porque ficaria um buraco no meio. O jeito certo é deslocar todos os contatos seguintes uma posição para trás, tampando o buraco, e depois diminuir a contagem.

#include <stdio.h>
#include <string.h>

#define MAX_CONTATOS 100
#define TAM_NOME 50
#define TAM_TEL 20

struct Contato {
    char nome[TAM_NOME];
    char telefone[TAM_TEL];
};

void tirar_novalinha(char *s) {
    s[strcspn(s, "\n")] = '\0';
}

int ler_linha(const char *rotulo, char *destino, int tam) {
    printf("%s", rotulo);
    if (fgets(destino, tam, stdin) == NULL) {
        return 0;
    }
    tirar_novalinha(destino);
    return 1;
}

void adicionar(struct Contato agenda[], int *n) {
    struct Contato c = {0};
    ler_linha("Nome: ", c.nome, TAM_NOME);
    ler_linha("Telefone: ", c.telefone, TAM_TEL);
    agenda[*n] = c;
    (*n)++;
    printf("Contato adicionado.\n");
}

void listar(const struct Contato agenda[], int n) {
    if (n == 0) {
        printf("Agenda vazia.\n");
        return;
    }
    printf("\n%-20s %s\n", "Nome", "Telefone");
    for (int i = 0; i < n; i++) {
        printf("%-20s %s\n", agenda[i].nome, agenda[i].telefone);
    }
}

void remover(struct Contato agenda[], int *n) {
    char alvo[TAM_NOME];
    ler_linha("Remover (nome exato): ", alvo, TAM_NOME);
    for (int i = 0; i < *n; i++) {
        if (strcmp(agenda[i].nome, alvo) == 0) {
            for (int j = i; j < *n - 1; j++) {
                agenda[j] = agenda[j + 1];
            }
            (*n)--;
            printf("Contato removido.\n");
            return;
        }
    }
    printf("Contato \"%s\" não encontrado.\n", alvo);
}

int main(void) {
    struct Contato agenda[MAX_CONTATOS];
    int n = 0;
    char opcao[10];
    int rodando = 1;
    while (rodando) {
        printf("\n=== Agenda da Ana ===\n");

O programa não carrega o arquivo, e a agenda abre vazia

O que aconteceu: quase sempre o programa foi rodado de uma pasta diferente da do agenda.txt. O fopen("agenda.txt", ...) procura o arquivo na pasta em que o programa roda, não na pasta do código.

Como resolver: rode o programa de dentro da pasta onde está o agenda.txt (cd ~/c/agenda). No editor do guia, o arquivo é a aba agenda.txt e fica sempre junto.

O programa completo

A versão final, igual à versão 4. Compile com gcc -std=c23 -Wall -Wextra agenda.c -o agenda:

#include <stdio.h>
#include <string.h>
#include <ctype.h>

#define MAX_CONTATOS 100
#define TAM_NOME 50
#define TAM_TEL 20
#define ARQUIVO "agenda.txt"

struct Contato {
    char nome[TAM_NOME];
    char telefone[TAM_TEL];
};

void tirar_novalinha(char *s) {
    s[strcspn(s, "\n")] = '\0';
}

int ler_linha(const char *rotulo, char *destino, int tam) {
    printf("%s", rotulo);
    if (fgets(destino, tam, stdin) == NULL) {
        return 0;
    }
    tirar_novalinha(destino);
    return 1;
}

int contem_ignorando_caso(const char *texto, const char *parte) {
    char t[TAM_NOME];
    char p[TAM_NOME];
    int i;
    for (i = 0; texto[i] != '\0' && i < TAM_NOME - 1; i++) {
        t[i] = (char) tolower((unsigned char) texto[i]);
    }
    t[i] = '\0';
    for (i = 0; parte[i] != '\0' && i < TAM_NOME - 1; i++) {
        p[i] = (char) tolower((unsigned char) parte[i]);
    }
    p[i] = '\0';
    return strstr(t, p) != NULL;
}

void adicionar(struct Contato agenda[], int *n) {
    if (*n >= MAX_CONTATOS) {
        printf("Agenda cheia.\n");
        return;
    }
    struct Contato c = {0};
    ler_linha("Nome: ", c.nome, TAM_NOME);
    ler_linha("Telefone: ", c.telefone, TAM_TEL);
    agenda[*n] = c;
    (*n)++;
    printf("Contato adicionado.\n");
}

void listar(const struct Contato agenda[], int n) {
    if (n == 0) {
        printf("Agenda vazia.\n");
        return;
    }
    printf("\n%-20s %s\n", "Nome", "Telefone");
    for (int i = 0; i < n; i++) {
        printf("%-20s %s\n", agenda[i].nome, agenda[i].telefone);
    }
}

void buscar(const struct Contato agenda[], int n) {
    char alvo[TAM_NOME];
    ler_linha("Buscar: ", alvo, TAM_NOME);
    int achou = 0;
    for (int i = 0; i < n; i++) {
        if (contem_ignorando_caso(agenda[i].nome, alvo)) {
            printf("%-20s %s\n", agenda[i].nome, agenda[i].telefone);
            achou = 1;
        }
    }
    if (!achou) {
        printf("Nenhum contato com \"%s\".\n", alvo);
    }
}

void remover(struct Contato agenda[], int *n) {
    char alvo[TAM_NOME];
    ler_linha("Remover (nome exato): ", alvo, TAM_NOME);
    for (int i = 0; i < *n; i++) {
        if (strcmp(agenda[i].nome, alvo) == 0) {
            for (int j = i; j < *n - 1; j++) {
                agenda[j] = agenda[j + 1];
            }
            (*n)--;
            printf("Contato removido.\n");
            return;
        }
    }
    printf("Contato \"%s\" não encontrado.\n", alvo);
}

void salvar(const struct Contato agenda[], int n) {
    FILE *f = fopen(ARQUIVO, "w");
    if (f == NULL) {
        perror(ARQUIVO);
        return;
    }
    for (int i = 0; i < n; i++) {
        fprintf(f, "%s;%s\n", agenda[i].nome, agenda[i].telefone);
    }
    fclose(f);
}

void carregar(struct Contato agenda[], int *n) {
    FILE *f = fopen(ARQUIVO, "r");
    if (f == NULL) {
        return;
    }
    char linha[TAM_NOME + TAM_TEL];
    while (*n < MAX_CONTATOS && fgets(linha, sizeof linha, f) != NULL) {
        if (sscanf(linha, "%49[^;];%19[^\n]", agenda[*n].nome, agenda[*n].telefone) == 2) {
            (*n)++;
        }
    }
    fclose(f);
}

int main(void) {
    struct Contato agenda[MAX_CONTATOS];
    int n = 0;
    carregar(agenda, &n);

    char opcao[10];
    int rodando = 1;
    while (rodando) {
        printf("\n=== Agenda da Ana ===\n");
        printf("1. Adicionar  2. Listar  3. Buscar  4. Remover  0. Sair\n");
        printf("Opção: ");
        if (fgets(opcao, sizeof opcao, stdin) == NULL) {
            break;
        }
        switch (opcao[0]) {
            case '1': adicionar(agenda, &n); salvar(agenda, n); break;
            case '2': listar(agenda, n); break;
            case '3': buscar(agenda, n); break;
            case '4': remover(agenda, &n); salvar(agenda, n); break;
            case '0': rodando = 0; break;
            default: printf("Opção inválida.\n");
        }
    }
    printf("Até logo!\n");
    return 0;
}

O que o projeto juntou: structs, um array e um contador, funções com passagem por ponteiro, fgets para texto com espaço, tolower e strstr/strcmp para a busca, o deslocamento do array na remoção e os arquivos para a agenda durar.

Resumo

  • Uma struct junta os dados de um contato; um array de structs e um contador guardam a agenda.
  • Ler tudo com fgets (inclusive a opção do menu) aceita nomes com espaço e evita o \n perdido do scanf.
  • Buscar sem diferenciar maiúsculas: passar os textos por tolower e usar strstr.
  • Remover de um array é deslocar os seguintes para trás e diminuir a contagem.
  • Gravar com fprintf e ler com fgets + sscanf (com larguras) faz a agenda durar depois de fechar o programa.
